rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- Helpers ----------
    function signedIn() {
      return request.auth != null;
    }
    function myEmail() {
      return request.auth.token.email;
    }
    function isAdmin() {
      return signedIn() && myEmail() == 'aaferyat@gmail.com';
    }
    // Est-ce que je suis membre du planning {pid} ?
    function isMember(pid) {
      return signedIn() &&
        exists(/databases/$(database)/documents/plannings/$(pid)/membres/$(myEmail()));
    }
    // Suis-je le propriétaire du planning {pid} ?
    function isOwner(pid) {
      return signedIn() &&
        get(/databases/$(database)/documents/plannings/$(pid)).data.ownerEmail == myEmail();
    }

    // ---------- Requetes collectionGroup sur "membres" ----------
    // Necessaire pour :
    //  - "mes plannings partages" : collectionGroup(membres) where email == le mien
    //  - l'admin qui liste tous les membres
    // Les rules imbriquees plus bas ne couvrent PAS les requetes collectionGroup,
    // il faut cette regle dediee.
    match /{path=**}/membres/{email} {
      allow read: if signedIn();
    }

    // Requete collectionGroup sur "historique" (vue globale admin des modifs).
    match /{path=**}/historique/{doc} {
      allow read: if isAdmin();
    }

    // ---------- Profils utilisateurs ----------
    // Chacun lit/écrit le sien ; l'admin peut tout lire.
    match /users/{email} {
      allow read: if isAdmin() || (signedIn() && myEmail() == email);
      allow write: if signedIn() && myEmail() == email;
    }

    // ---------- Plannings ----------
    match /plannings/{pid} {
      // Lecture : propriétaire, membre, ou admin.
      // (les requêtes "mes plannings" filtrent par ownerEmail == mon email,
      //  les "partagés" passent par la sous-collection membres)
      allow read: if isAdmin()
        || (signedIn() && resource.data.ownerEmail == myEmail())
        || isMember(pid);

      // Création : connecté, et je me déclare propriétaire.
      allow create: if signedIn() && request.resource.data.ownerEmail == myEmail();

      // Modification : propriétaire, membre (éditeur), ou admin.
      allow update: if isAdmin() || isOwner(pid) || isMember(pid);

      // Suppression : propriétaire ou admin uniquement.
      allow delete: if isAdmin() || isOwner(pid);

      // --- Sous-collection membres ---
      match /membres/{email} {
        // Lecture : tout connecté (nécessaire pour la requête collectionGroup
        // "mes plannings partagés" filtrée par email == le mien) + admin.
        allow read: if signedIn();
        // Ajout de soi-même (rejoindre via invitation) OU géré par le propriétaire/admin.
        allow create: if signedIn() && (email == myEmail() || isOwner(pid) || isAdmin());
        // Modifier/retirer un membre : propriétaire, admin, ou soi-même.
        allow update, delete: if isAdmin() || isOwner(pid) || (signedIn() && email == myEmail());
      }

      // --- Sous-collection historique (traçabilité) ---
      match /historique/{doc} {
        allow read: if isAdmin() || isOwner(pid) || isMember(pid);
        allow create: if isAdmin() || isOwner(pid) || isMember(pid);
        allow update: if false; // l'historique ne se modifie pas
        allow delete: if isAdmin(); // seul l'admin peut purger
      }
    }

    // ---------- Invitations ----------
    // Lecture : tout connecté (nécessaire pour résoudre un code et rejoindre).
    // Création/suppression : connecté (le créateur). Mise à jour usedBy au join.
    match /invitations/{code} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.createdBy == myEmail();
      allow update: if signedIn(); // usedBy mis à jour quand quelqu'un rejoint
      allow delete: if isAdmin() || (signedIn() && resource.data.createdBy == myEmail());
    }

    // ---------- Journal d'événements (admin) ----------
    // Tout connecté peut écrire (login, join, create...). Lecture : admin seul.
    match /events/{id} {
      allow create: if signedIn();
      allow read: if isAdmin();
      allow update: if false;
      allow delete: if isAdmin();
    }
  }
}
